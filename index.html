Implementation Details

This calculator uses modern web technologies for optimal performance and flexibility:

Why This Approach?

  • Pure JavaScript - No external dependencies or libraries required
  • CSS Grid Layout - Perfectly aligned buttons with minimal code
  • Responsive Design - Works on all device sizes
  • Keyboard Support - Full keyboard accessibility
  • Memory Functions - MR, MC, M+, M- operations

Technical Breakdown

  1. Display System: Dual display showing both current input and calculation history
  2. Math Evaluation: Uses JavaScript's eval() with safety checks and expression translation
  3. Scientific Functions: Implements trigonometry, logarithms, factorials, and constants
  4. Theme Switching: Light/dark mode toggle with CSS variables
  5. Error Handling: Graceful error recovery and user feedback

    <script>
        // Calculator state
        let currentInput = '0';
        let calculation = '';
        let memory = 0;
        let isDarkTheme = false;

        // DOM elements
        const currentDisplay = document.getElementById('current');
        const historyDisplay = document.getElementById('history');

        // Update display
        function updateDisplay() {
            currentDisplay.textContent = currentInput;
            historyDisplay.textContent = calculation;
        }

        // Handle number and operator inputs
        function calculate(value) {
            if (value === '=') {
                try {
                    // Replace display symbols with JavaScript operators
                    let expression = calculation + currentInput;
                    expression = expression.replace(/×/g, '*').replace(/÷/g, '/');
                    
                    // Handle percentage calculations
                    expression = expression.replace(/(\d+)%/g, '($1/100)');
                    
                    // Evaluate the expression
                    const result = eval(expression);
                    currentInput = result.toString();
                    calculation = '';
                } catch (error) {
                    currentInput = 'Error';
                    calculation = '';
                }
            } else if (/[+\-*/^]/.test(value)) {
                // Handle operators
                calculation = calculation + currentInput + value;
                currentInput = '0';
            } else if (value === 'π') {
                currentInput = Math.PI.toString();
            } else if (value === 'e') {
                currentInput = Math.E.toString();
            } else if (currentInput === '0' || currentInput === 'Error') {
                currentInput = value;
            } else {
                currentInput += value;
            }
            
            updateDisplay();
        }

        // Clear all
        function clearAll() {
            currentInput = '0';
            calculation = '';
            updateDisplay();
        }

        // Backspace
        function backspace() {
            if (currentInput.length === 1) {
                currentInput = '0';
            } else {
                currentInput = currentInput.slice(0, -1);
            }
            updateDisplay();
        }

        // Toggle sign
        function toggleSign() {
            if (currentInput.startsWith('-')) {
                currentInput = currentInput.substring(1);
            } else {
                currentInput = '-' + currentInput;
            }
            updateDisplay();
        }

        // Factorial
        function factorial() {
            let num = parseFloat(currentInput);
            if (num === 0 || num === 1) {
                currentInput = '1';
            } else {
                let result = 1;
                for (let i = 2; i <= num; i++) {
                    result *= i;
                }
                currentInput = result.toString();
            }
            updateDisplay();
        }

        // Memory functions
        function memoryRecall() {
            currentInput = memory.toString();
            updateDisplay();
        }

        function memoryClear() {
            memory = 0;
        }

        function memoryAdd() {
            memory += parseFloat(currentInput);
        }

        function memorySubtract() {
            memory -= parseFloat(currentInput);
        }

        // Toggle theme
        function toggleTheme() {
            isDarkTheme = !isDarkTheme;
            document.body.classList.toggle('dark-theme');
            
            // Update theme button icon
            const themeIcon = document.querySelector('.theme-btn i');
            themeIcon.className = isDarkTheme ? 'fas fa-sun' : 'fas fa-moon';
        }

        // Initialize calculator
        updateDisplay();
    </script>
</body>
</html>
